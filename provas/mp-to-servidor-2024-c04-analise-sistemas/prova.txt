64 O trecho de código a seguir exemplifica um polimorfismo
   em Java, pois a classe Gerente implementa de maneira
   específica o método recebeSalario, ao mesmo tempo em
   que compartilha parte da funcionalidade da classe
   Empregado.

     class Empregado {
       private String nome;
       private double salario;
       public Empregado(String nome, double
            salario) {
            this.nome = nome;
            this.salario = salario;
       }
       public double recebeSalario() {
            return salario;
       }
     }
     class Gerente extends Empregado{
        private double bonus;

       public Gerente(String nome, double
               salario, double bonus) {
           super(nome, salario);
           this.bonus = bonus;
        }
     @Override
        public double recebeSalario() {
         return super.recebeSalario() + bonus;
        }
     }
     class FolhaPagamento {
         public static void main(String[] args) {
         Gerente gerente = new Gerente(“Luiza”,
              5000, 1000);
     System.out.println(gerente.recebeSalario());
     }
     }
